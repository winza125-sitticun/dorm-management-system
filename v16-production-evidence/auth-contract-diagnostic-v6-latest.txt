214:function normalizeRoomName(name: string): string {
215:  if (!name) return "";
216:  let clean = name.toString().trim().toLowerCase()
217:    .replace(/^ห้อง\s*/, "")
218:    .replace(/\s+/g, "");
219:  
220:  const thaiNumberMap: Record<string, string> = {
221:    "แรก": "1", "หนึ่ง": "1", "สอง": "2", "สาม": "3", "สี่": "4", "ห้า": "5",
222:    "หก": "6", "เจ็ด": "7", "แปด": "8", "เก้า": "9", "สิบ": "10"
223:  };
224:
225:  for (const [thaiWord, digit] of Object.entries(thaiNumberMap)) {
226:    if (clean === thaiWord) {
227:      clean = digit;
228:    } else {
229:      clean = clean.replace(new RegExp(thaiWord, 'g'), digit);
230:    }
231:  }
232:  
233:  return clean;
234:}
